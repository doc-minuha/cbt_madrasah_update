function saveDB() {
    localStorage.setItem(DB_KEY, JSON.stringify(db));
    clearTimeout(syncTimeout);
    syncTimeout = setTimeout(() => {
        syncToCloud();
    }, 3000); 
}

function syncToCloud() {
    const url = getScriptUrl();
    if (!url) return;
    fetch(url, {
        method: 'POST',
        body: JSON.stringify(db),
        headers: { 'Content-Type': 'text/plain;charset=utf-8' }
    }).catch(err => {
        console.warn("Sinkronisasi cloud latar belakang tertunda, data aman di lokal:", err && err.message ? err.message : err);
    });
}

// FUNGSI: Khusus untuk siswa mengirim hasil ujian (Aman dari tabrakan data)
function syncHasilSiswaToCloud(dataHasilSiswa) {
    const url = getScriptUrl();
    if (!url) return;
    let payload = {
        action: "simpan_nilai_siswa",
        data: dataHasilSiswa
    };

    fetch(url, {
        method: 'POST',
        body: JSON.stringify(payload),
        headers: { 'Content-Type': 'text/plain;charset=utf-8' }
    })
    .then(res => res.json())
    .then(res => {
        console.log("Hasil ujian berhasil dikirim ke Cloud:", res);
    })
    .catch(err => {
        console.warn("Kirim hasil ujian ke Cloud tertunda, data aman di lokal:", err && err.message ? err.message : err);
    });
}

// FUNGSI LOAD DATABASE (Local-First, aman offline & cloud sync)
async function loadDB() {
    // LANGKAH 1: Ambil data dari localStorage SECARA SINKRONUS (INSTAN)
    const localData = localStorage.getItem(DB_KEY);
    if (localData) {
        try {
            let parsed = JSON.parse(localData);
            db = { ...db, ...parsed };
        } catch(e) {
            console.warn("Gagal membaca cache lokal:", e);
        }
    }

    // Tampilkan UI awal menggunakan data lokal terlebih dahulu
    updateGatewayInfo();
    updateLoginMethodUI();
    refreshLoginDropdowns();

    const cloudUrl = getScriptUrl();
    if (!cloudUrl) {
        // Mode lokal penuh (tanpa cloud script URL)
        return;
    }

    // LANGKAH 2: Ambil data terbaru dari Cloud jika URL dikonfigurasi
    const isFirstTime = !localData || (Array.isArray(db.groups) && db.groups.length === 0);
    if (isFirstTime) {
        Swal.fire({
            title: 'Menghubungkan Server...',
            text: 'Mengunduh data ujian dari Cloud...',
            allowOutsideClick: false,
            didOpen: () => { Swal.showLoading(); }
        });
    }

    try {
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 8000);
        const response = await fetch(cloudUrl, { signal: controller.signal });
        clearTimeout(timeoutId);

        if (response.ok) {
            const cloudData = await response.json();
            if (cloudData && typeof cloudData === 'object' && cloudData.settings) {
                db = { ...db, ...cloudData };
                localStorage.setItem(DB_KEY, JSON.stringify(db));
                
                updateGatewayInfo();
                updateLoginMethodUI();
                refreshLoginDropdowns();
            }
        }
